async function seedQA() {
  const topics = await prisma.topic.findMany();
  const bySlug = new Map(topics.map((t) => [t.slug, t]));
  const seedUser = await prisma.user.findUnique({ where: { email: "seed@theceylonx.com" } });

  let qCount = 0;
  let aCount = 0;

  for (const item of QA) {
    const topic = bySlug.get(item.topicSlug);
    if (!topic) continue;

    const slug = slugify(item.title);
    // Skip if already exists
    const existing = await prisma.question.findUnique({ where: { slug } });
    let questionId: string;

    if (!existing) {
      const q = await prisma.question.create({
        data: {
          title: item.title,
          body: item.bodyHtml,
          tags: item.tags.map((t) => t.toLowerCase()),
          slug,
          topicId: topic.id,
          user: { connect: { id: seedUser!.id } },
          votesCount: item.votes ?? 0,
        },
      });
      questionId = q.id;
      qCount++;
    } else {
      questionId = existing.id;
    }

    // Answers
    let acceptedAnswerId: string | null = null;
    for (const ans of item.answers) {
      const user = await prisma.user.findUnique({ where: { email: ans.authorEmail } });
      const a = await prisma.answer.create({
        data: {
          body: ans.bodyHtml,
          questionId,
          userId: user ? user.id : seedUser!.id,
          votesCount: ans.votes ?? 0,
          isAccepted: !!ans.accepted,
        },
      });
      if (ans.accepted) acceptedAnswerId = a.id;
      aCount++;
    }
    if (acceptedAnswerId) {
      await prisma.question.update({
        where: { id: questionId },
        data: { acceptedAnswerId },
      });
    }
    // Update answersCount
    const count = await prisma.answer.count({ where: { questionId } });
    await prisma.question.update({ where: { id: questionId }, data: { answersCount: count } });
  }

  console.log(`Seeded Qs: ${qCount}, As: ${aCount} ✅`);
}

async function main() {
  await seedTopics();
  await seedUsers();
  await seedQA();
}

main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(async () => prisma.$disconnect());

4) Image selection service
Create server/services/imageSelectorService.ts:
import { CATEGORY_CHOICES, TripCategory } from './categoryImageMap';
import { createHash } from 'crypto';

export function getCuratedChoices(category: TripCategory) {
  return CATEGORY_CHOICES[category] ?? CATEGORY_CHOICES.unknown;
}

export function pickDefaultFromChoices(category: TripCategory, seed: string) {
  const choices = getCuratedChoices(category);
  const idx = Number('0x' + createHash('md5').update(seed).digest('hex').slice(0,4)) % choices.length;
  const chosen = choices[idx];
  return {
    imageUrl: chosen.url,
    imageProvider: 'curated' as const,
    imageAttribution: chosen.attribution ?? null,
    imageFetchedAt: new Date()
  };
}

export function isAllowedCategoryImage(category: TripCategory, url: string) {
  return getCuratedChoices(category).some(c => c.url === url);
}

After validation:

Compute default via pickDefaultFromChoices(category, seed = title + organizerId).

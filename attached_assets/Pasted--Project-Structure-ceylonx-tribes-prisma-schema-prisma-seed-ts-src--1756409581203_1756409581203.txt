## src/app/tribes/page.tsx
```tsx
import { prisma } from "@/src/lib/db";
import TopicFilterPills from "@/src/components/TopicFilterPills";
import SearchBar from "@/src/components/SearchBar";
import SortTabs from "@/src/components/SortTabs";
import QuestionCard from "@/src/components/QuestionCard";

function TribesHero() {
  return (
    <section className="mb-6 rounded-2xl border bg-gradient-to-br from-emerald-50 to-teal-50 p-6 md:p-8">
      <h1 className="text-3xl md:text-4xl font-extrabold tracking-tight">Welcome to CeylonX Tribes</h1>
      <p className="mt-2 max-w-2xl text-neutral-700">
        A friendly, English-only space to ask questions, share tips, and uncover Sri Lanka’s hidden gems. Vote on the best
        answers, mark solutions, and help travelers expand their journey.
      </p>
      <div className="mt-4 flex flex-wrap items-center gap-3 text-sm">
        <a href="#ask" className="rounded-xl bg-black text-white px-4 py-2">Ask a Question</a>
        <a href="/tribes?sort=unanswered" className="rounded-xl border px-4 py-2">Answer an Unanswered</a>
      </div>
    </section>
  );
}

export const dynamic = "force-dynamic";

export default async function TribesPage({ searchParams }: { searchParams: Record<string, string | string[] | undefined> }) {
  const q = (searchParams.q as string) || "";
  const topic = (searchParams.topic as string) || "";
  const sort = (searchParams.sort as string) || "top"; // top | new | unanswered

  const where: any = {
    AND: [
      q
        ? {
            OR: [
              { title: { contains: q, mode: "insensitive" } },
              { body: { contains: q, mode: "insensitive" } },
              { tags: { has: q.toLowerCase() } },
            ],
          }
        : {},
      topic ? { topic: { slug: topic } } : {},
    ],
  };

  const orderBy =
    sort === "new"
      ? { createdAt: "desc" as const }
      : sort === "unanswered"
      ? [{ answersCount: "asc" as const }, { createdAt: "desc" as const }]
      : { votesCount: "desc" as const };

  const [topics, questions] = await Promise.all([
    prisma.topic.findMany({ orderBy: { name: "asc" } }),
    prisma.question.findMany({
      where,
      orderBy,
      include: { topic: true, user: true },
      take: 20,
    }),
  ]);

  return (
    <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
      <section className="lg:col-span-9 space-y-4">
        <TribesHero />
        <div className="flex flex-col md:flex-row gap-3">
          <SearchBar defaultValue={q} />
          <SortTabs current={sort} />
        </div>
        <TopicFilterPills topics={topics} current={topic} />
        <div className="space-y-3">
          {questions.map((qu) => (
            <QuestionCard key={qu.id} q={qu} />
          ))}
          {questions.length === 0 && (
            <div className="text-sm text-neutral-500 p-6 border rounded-xl">No results. Try different keywords or topics.</div>
          )}
        </div>
      </section>
      <aside className="lg:col-span-3 space-y-4">
        <div className="p-4 border rounded-2xl bg-white">
          <h3 className="font-semibold mb-2">Topics</h3>
          <ul className="text-sm space-y-1">
            {topics.map((t) => (
              <li key={t.id}><a className={`hover:underline ${t.slug === topic ? "font-semibold" : ""}`} href={`?topic=${t.slug}`}>{t.name}</a></li>
            ))}
          </ul>
        </div>
        <div className="p-4 border rounded-2xl bg-white">
          <h3 className="font-semibold mb-2">Tips</h3>
          <p className="text-sm text-neutral-600">Use keywords like <em>Ayurveda</em>, <em>Knuckles</em>, <em>homestay</em>, <em>surf</em> or filter by topic to find answers faster.</p>
        </div>
      </aside>
    </div>
  );
}
